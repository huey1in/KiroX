cmake_minimum_required(VERSION 3.25)
project(KiroX VERSION 2.0.0 LANGUAGES CXX)
if(WIN32)
  enable_language(RC)
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
option(KIROX_BUILD_DESKTOP "Build the Qt Quick desktop application" ON)
option(KIROX_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)

find_package(Qt6 6.8 REQUIRED COMPONENTS Core Network)
qt_standard_project_setup(REQUIRES 6.8)
add_library(kirox_options INTERFACE)
target_compile_features(kirox_options INTERFACE cxx_std_20)
if(MSVC)
  target_compile_options(kirox_options INTERFACE /W4 /permissive- /utf-8)
  if(KIROX_WARNINGS_AS_ERRORS)
    target_compile_options(kirox_options INTERFACE /WX)
  endif()
else()
  target_compile_options(kirox_options INTERFACE -Wall -Wextra -Wpedantic)
  if(KIROX_WARNINGS_AS_ERRORS)
    target_compile_options(kirox_options INTERFACE -Werror)
  endif()
endif()

include(cmake/CurlImpersonate.cmake)
add_subdirectory(src/domain)
add_subdirectory(src/application)
add_subdirectory(src/infrastructure)
if(KIROX_BUILD_DESKTOP)
  find_package(Qt6 6.8 REQUIRED COMPONENTS Gui Quick QuickControls2 Svg Widgets Multimedia)
  add_subdirectory(src/desktop)
endif()
if(BUILD_TESTING)
  find_package(Qt6 6.8 REQUIRED COMPONENTS Test)
  add_subdirectory(tests)
endif()

set(CPACK_PACKAGE_NAME "KiroX")
set(CPACK_PACKAGE_VENDOR "huey1in")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_RESOURCE_FILE_LICENSE "${PROJECT_SOURCE_DIR}/LICENSE")
install(FILES LICENSE licenses/THIRD-PARTY.md DESTINATION licenses)
install(DIRECTORY licenses/qt DESTINATION licenses)
install(DIRECTORY licenses/mingw licenses/openssl DESTINATION licenses)
include(CPack)
